27: static inline bool checkCudaRuntime(cudaError_t error, std::string const &message = "") {
28:   if (error != cudaSuccess) {
29:     logger::error("{} CUDA Error: {}", message, cudaGetErrorName(error));
30:     return false;
31:   }
32:   return true;
33: }

35: bool DenoiserOidn::init(bool albedo, bool normal, bool hdr) {
36:   int device;
37:   if (!checkCudaRuntime(cudaGetDevice(&device))) {
38:     return false;
39:   }
40:   if (!checkCudaRuntime(cudaStreamCreate(&mCudaStream))) {
41:     return false;
42:   }
43:   mDevice = oidn::newCUDADevice(device, mCudaStream);
44:   mDevice.commit();
45: 
46:   mCommandPool = core::Context::Get()->createCommandPool();
47:   mCommandBufferIn = mCommandPool->allocateCommandBuffer();
48:   mCommandBufferOut = mCommandPool->allocateCommandBuffer();
49: 
50:   mPixelSize = 16;
51: 
52:   mAlbedo = albedo;
53:   mNormal = normal;
54:   mHdr = hdr;
55: 
56:   return true;
57: }
58: 
59: void DenoiserOidn::allocate(uint32_t width, uint32_t height) {
60:   mWidth = width;
61:   mHeight = height;
62:   free();
63: 
64:   mFilter = mDevice.newFilter("RT");
65: 
66:   mInputBuffer = core::Buffer::Create(width * height * mPixelSize,
67:                                       vk::BufferUsageFlagBits::eTransferDst |
68:                                           vk::BufferUsageFlagBits::eTransferSrc,
69:                                       VMA_MEMORY_USAGE_GPU_ONLY, VmaAllocationCreateFlags{}, true);
70:   mInputBufferOidn = mDevice.newBuffer(mInputBuffer->getCudaPtr(), mInputBuffer->getSize());
71:   mFilter.setImage("color", mInputBufferOidn, oidn::Format::Float3, mWidth, mHeight, 0, 16);
72: 
73:   mFilter.setImage("output", mInputBufferOidn, oidn::Format::Float3, mWidth, mHeight, 0, 16);
74: 
75:   if (useAlbedo()) {
76:     mAlbedoBuffer = core::Buffer::Create(
77:         width * height * mPixelSize,
78:         vk::BufferUsageFlagBits::eTransferDst | vk::BufferUsageFlagBits::eTransferSrc,
79:         VMA_MEMORY_USAGE_GPU_ONLY, VmaAllocationCreateFlags{}, true);
80:     mAlbedoBufferOidn = mDevice.newBuffer(mAlbedoBuffer->getCudaPtr(), mAlbedoBuffer->getSize());
81:     mFilter.setImage("albedo", mAlbedoBufferOidn, oidn::Format::Float3, mWidth, mHeight, 0, 16);
82:   }
83:   if (useNormal()) {
84:     mNormalBuffer = core::Buffer::Create(
85:         width * height * mPixelSize,
86:         vk::BufferUsageFlagBits::eTransferDst | vk::BufferUsageFlagBits::eTransferSrc,
87:         VMA_MEMORY_USAGE_GPU_ONLY, VmaAllocationCreateFlags{}, true);
88:     mNormalBufferOidn = mDevice.newBuffer(mNormalBuffer->getCudaPtr(), mNormalBuffer->getSize());
89:     mFilter.setImage("normal", mNormalBufferOidn, oidn::Format::Float3, mWidth, mHeight, 0, 16);
90:   }
91:   mFilter.set("hdr", mHdr);
92:   mFilter.commit();
93: 
94:   vk::Device device = core::Context::Get()->getDevice();
95:   // create timeline semaphore with value 0
96:   vk::SemaphoreTypeCreateInfo timelineCreateInfo(vk::SemaphoreType::eTimeline, 0);
97:   vk::SemaphoreCreateInfo createInfo{};
98:   vk::ExportSemaphoreCreateInfo exportCreateInfo(
99:       vk::ExternalSemaphoreHandleTypeFlagBits::eOpaqueFd);
100:   createInfo.setPNext(&exportCreateInfo);
101:   exportCreateInfo.setPNext(&timelineCreateInfo);
102:   mSem = device.createSemaphoreUnique(createInfo);
103: 
104:   // export vulkan semaphore as cuda semaphore
105:   int fd =
106:       device.getSemaphoreFdKHR({mSem.get(), vk::ExternalSemaphoreHandleTypeFlagBits::eOpaqueFd});
107:   cudaExternalSemaphoreHandleDesc desc = {};
108:   desc.flags = 0;
109:   desc.handle.fd = fd;
110:   desc.type = cudaExternalSemaphoreHandleTypeTimelineSemaphoreFd;
111:   checkCudaRuntime(cudaImportExternalSemaphore(&mCudaSem, &desc));
112: }

114: void DenoiserOidn::free() {
115:   core::Context::Get()->getDevice().waitIdle();
116:   if (mCudaSem) {
117:     cudaDestroyExternalSemaphore(mCudaSem);
118:     mCudaSem = {};
119:   }
120: 
121:   mFilter = {};
122:   mNormalBufferOidn = {};
123:   mNormalBuffer = {};
124:   mAlbedoBufferOidn = {};
125:   mAlbedoBuffer = {};
126:   mInputBufferOidn = {};
127:   mInputBuffer = {};
128: }

130: void DenoiserOidn::denoise(core::Image &color, core::Image *albedo, core::Image *normal) {
131: 
132:   if (color.getFormat() != vk::Format::eR32G32B32A32Sfloat ||
133:       (albedo && albedo->getFormat() != vk::Format::eR32G32B32A32Sfloat) ||
134:       (normal && normal->getFormat() != vk::Format::eR32G32B32A32Sfloat)) {
135:     throw std::runtime_error("denoiser only supports R32G32B32A32Sfloat format");
136:   }
137: 
138:   // copy external image to buffer
139:   mCommandBufferIn->reset();
140:   mCommandBufferIn->begin({vk::CommandBufferUsageFlagBits::eOneTimeSubmit});
141: 
142:   color.recordCopyToBuffer(mCommandBufferIn.get(), mInputBuffer->getVulkanBuffer(), 0,
143:                            mInputBuffer->getSize(), {0, 0, 0}, color.getExtent());
144: 
145:   if (useAlbedo() && albedo) {
146:     albedo->recordCopyToBuffer(mCommandBufferIn.get(), mAlbedoBuffer->getVulkanBuffer(), 0,
147:                                mAlbedoBuffer->getSize(), {0, 0, 0}, color.getExtent());
148:   }
149:   if (useNormal() && normal) {
150:     normal->recordCopyToBuffer(mCommandBufferIn.get(), mNormalBuffer->getVulkanBuffer(), 0,
151:                                mNormalBuffer->getSize(), {0, 0, 0}, color.getExtent());
152:   }
153: 
154:   mCommandBufferIn->end();
155: 
156:   core::Context::Get()->getQueue().submit(mCommandBufferIn.get(), {}, {}, {}, mSem.get(),
157:                                           ++mSemValue, {});
158: 
159:   cudaExternalSemaphoreWaitParams waitParams{};
160:   waitParams.params.fence.value = mSemValue;
161:   cudaWaitExternalSemaphoresAsync(&mCudaSem, &waitParams, 1, mCudaStream);
162: 
163:   // run
164:   // mFilter.executeAsync();
165:   mFilter.execute();
166: 
167:   char const *err;
168:   if (mDevice.getError(err) != oidn::Error::None) {
169:     logger::error("OIDN Error: {}", err);
170:   }
171: 
172:   // copy output buffer to external image
173:   cudaExternalSemaphoreSignalParams sigParams{};
174:   sigParams.flags = 0;
175:   sigParams.params.fence.value = ++mSemValue;
176:   cudaSignalExternalSemaphoresAsync(&mCudaSem, &sigParams, 1, mCudaStream);
177: 
178:   mCommandBufferOut->reset();
179:   mCommandBufferOut->begin({vk::CommandBufferUsageFlagBits::eOneTimeSubmit});
180:   color.recordCopyFromBuffer(mCommandBufferOut.get(), mInputBuffer->getVulkanBuffer(), 0,
181:                              mInputBuffer->getSize(), {0, 0, 0}, color.getExtent());
182:   mCommandBufferOut->end();
183: 
184:   vk::PipelineStageFlags waitStage = vk::PipelineStageFlagBits::eTransfer;
185:   core::Context::Get()->getQueue().submit(mCommandBufferOut.get(), mSem.get(), waitStage,
186:                                           mSemValue, {}, {}, {});
187: }
188: 
189: DenoiserOidn::~DenoiserOidn() {
190:   logger::info("OIDN finished");
191:   free();
192: 
193:   mDevice = {};
194:   if (mCudaStream) {
195:     checkCudaRuntime(cudaStreamDestroy(mCudaStream));
196:   }
197: }
