def verdict(note):
    if note.startswith(('弱','形态清楚但语义弱')):return 'weak'
    if '混杂强' in note or note.startswith(('谨慎','失败例谨慎')):return 'confounded'
    if '阶段差异' in note and '候选' not in note:return 'phase_only'
    if '候选' in note:return 'candidate'
    return 'confounded'
